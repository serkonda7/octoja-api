Usage

import { createOctojaClient } from 'octoja-api'

const octoja = createOctojaClient({ baseUrl: 'https://acme.eu1.octoja.cloud' })
const { refreshToken } = await octoja.auth.login({
  mailAddress: process.env.OCTOJA_USER!,
  password: process.env.OCTOJA_PASS!,
}) // access token is stored on the client

const customers = await octoja.customers.list({ search: 'acme' })
await octoja.auth.refresh(refreshToken) // rotate without rebuilding

createOctojaClient(options)

Isolated hey-api instance (no shared client singleton, so agent tokens cannot leak into technician requests).

interface CreateOctojaClientOptions {
/**
	 * Base URL of the Octoja instance, e.g. `https://acme.eu1.octoja.cloud`.
	 */
	baseUrl: string
	/**
	 * Access token, sent as `Authorization: Bearer <token>`. Optional so a
	 * client can be created first and authenticated later via `auth.login`
	 * or `setToken`.
	 */
	token?: string
	/**
	 * Custom fetch implementation. Defaults to `globalThis.fetch`.
	 */
	fetch?: typeof fetch
	/**
	 * Extra hey-api client config merged over the defaults.
	 */
	config?: Omit<Config, 'baseUrl' | 'auth' | 'fetch'>
}
